from flask import Flask, jsonify, request, Response
from datetime import datetime


# ============================================================
# DUQUE - SERVIDOR CENTRAL
# ============================================================

app = Flask(__name__)


# ============================================================
# ESTADO ATUAL DO DUQUE
# ============================================================

estado_duque = {
    "estado": "standby",
    "tarefa": "",
    "atividade": "Sistema online",
    "coerencia": 100,
    "ultima_atualizacao": None,
}


# ============================================================
# ESTADOS PERMITIDOS
# ============================================================

ESTADOS_PERMITIDOS = [
    "standby",
    "ouvindo",
    "processando",
    "executando",
    "falando",
]


# ============================================================
# PÁGINA PRINCIPAL
# ============================================================

@app.route("/")
def inicio():

    caminho = "interface/index.html"

    with open(
        caminho,
        "r",
        encoding="utf-8"
    ) as arquivo:

        html = arquivo.read()

    # --------------------------------------------------------
    # Injeta automaticamente a ponte entre a HUD e o servidor.
    #
    # O index.html original não precisa ser alterado.
    # --------------------------------------------------------

    ponte = """
<script>
/* =========================================================
   DUQUE — BRIDGE SERVER → HUD
   ========================================================= */

(function(){

    const URL_ESTADO = "/api/estado";

    let ultimoEstado = null;
    let ultimaTarefa = null;
    let ultimaAtividade = null;


    /* -------------------------------------------------------
       Converter estado do servidor para estado visual da HUD
       ------------------------------------------------------- */

    function sincronizarEstado(dados){

        if(!dados){
            return;
        }

        if(typeof window.DUQUE === "undefined"){
            return;
        }


        const estado = dados.estado || "standby";

        const tarefa = dados.tarefa || "";

        const atividade = dados.atividade || "";

        const coerencia = (
            typeof dados.coerencia === "number"
            ? dados.coerencia
            : null
        );


        /* ---------------------------------------------------
           Atualizar estado principal
           --------------------------------------------------- */

        const patch = {
            state: estado,
            currentTask: tarefa
        };


        /*
         * Não deixamos o relógio/clima da HUD serem
         * sobrescritos pelo servidor.
         */

        window.DUQUE.setState(
            patch
        );


        /* ---------------------------------------------------
           Atividade
           --------------------------------------------------- */

        if(
            atividade &&
            atividade !== ultimaAtividade
        ){

            ultimaAtividade = atividade;

            if(
                typeof window.DUQUE.activity === "function"
            ){

                window.DUQUE.activity(
                    atividade
                );
            }
        }


        /* ---------------------------------------------------
           Coerência
           --------------------------------------------------- */

        if(coerencia !== null){

            const el = document.getElementById(
                "coherence"
            );

            if(el){

                el.dataset.serverCoherence =
                    coerencia.toFixed(3);
            }
        }


        /* ---------------------------------------------------
           Telemetria de mudança de estado
           --------------------------------------------------- */

        if(
            ultimoEstado !== null &&
            ultimoEstado !== estado
        ){

            if(
                typeof window.DUQUE.telemetry === "function"
            ){

                const nomes = {
                    standby: "Standby",
                    ouvindo: "Canal de voz",
                    processando: "Processamento",
                    executando: "Computer Use",
                    falando: "Síntese de voz"
                };

                window.DUQUE.telemetry(
                    [
                        {
                            k: "Estado",
                            v:
                                nomes[estado]
                                || estado
                        }
                    ],
                    2200
                );
            }
        }


        ultimoEstado = estado;
    }


    /* -------------------------------------------------------
       Consultar servidor
       ------------------------------------------------------- */

    async function atualizar(){

        try{

            const resposta = await fetch(
                URL_ESTADO,
                {
                    method: "GET",
                    cache: "no-store"
                }
            );


            if(!resposta.ok){

                return;
            }


            const dados =
                await resposta.json();


            sincronizarEstado(
                dados
            );

        }catch(erro){

            /*
             * O servidor pode estar temporariamente
             * indisponível durante reinicializações.
             *
             * Não exibimos erro na HUD para não poluir
             * a interface.
             */

        }
    }


    /* -------------------------------------------------------
       Primeira sincronização
       ------------------------------------------------------- */

    atualizar();


    /* -------------------------------------------------------
       Sincronização contínua
       ------------------------------------------------------- */

    setInterval(
        atualizar,
        300
    );


    console.log(
        "[DUQUE] HUD conectada ao servidor."
    );

})();
</script>
"""


    # --------------------------------------------------------
    # Inserir antes do fechamento do BODY
    # --------------------------------------------------------

    marcador = "</body>"

    if marcador in html:

        html = html.replace(
            marcador,
            ponte + "\n" + marcador,
            1
        )

    else:

        html += ponte


    return Response(
        html,
        mimetype="text/html"
    )


# ============================================================
# API - OBTER ESTADO
# ============================================================

@app.route(
    "/api/estado",
    methods=["GET"]
)
def obter_estado():

    return jsonify(
        estado_duque
    )


# ============================================================
# API - ALTERAR ESTADO
# ============================================================

@app.route(
    "/api/estado",
    methods=["POST"]
)
def alterar_estado():

    dados = request.get_json(
        silent=True
    )


    if not dados:

        return jsonify({
            "erro":
                "JSON não informado."
        }), 400


    novo_estado = dados.get(
        "estado"
    )


    if novo_estado not in ESTADOS_PERMITIDOS:

        return jsonify({
            "erro":
                f"Estado inválido: {novo_estado}",

            "estados_permitidos":
                ESTADOS_PERMITIDOS

        }), 400


    # --------------------------------------------------------
    # Estado
    # --------------------------------------------------------

    estado_duque["estado"] = (
        novo_estado
    )


    # --------------------------------------------------------
    # Tarefa
    # --------------------------------------------------------

    if "tarefa" in dados:

        estado_duque["tarefa"] = (
            dados["tarefa"] or ""
        )


    # --------------------------------------------------------
    # Atividade
    # --------------------------------------------------------

    if "atividade" in dados:

        estado_duque["atividade"] = (
            dados["atividade"] or ""
        )


    # --------------------------------------------------------
    # Coerência
    # --------------------------------------------------------

    if "coerencia" in dados:

        try:

            estado_duque["coerencia"] = max(
                0,
                min(
                    100,
                    int(
                        dados["coerencia"]
                    )
                )
            )

        except (
            TypeError,
            ValueError
        ):

            pass


    # --------------------------------------------------------
    # Timestamp
    # --------------------------------------------------------

    estado_duque[
        "ultima_atualizacao"
    ] = datetime.now().isoformat(
        timespec="milliseconds"
    )


    # --------------------------------------------------------
    # Log
    # --------------------------------------------------------

    print(
        "[DUQUE] "
        f"{novo_estado.upper()} | "
        f"{estado_duque['atividade']} | "
        f"{estado_duque['tarefa']}"
    )


    return jsonify(
        estado_duque
    )


# ============================================================
# COMPATIBILIDADE COM ENDPOINT ANTIGO
# ============================================================

@app.route(
    "/estado/<novo_estado>",
    methods=["GET"]
)
def estado_compatibilidade(
    novo_estado
):

    if novo_estado not in ESTADOS_PERMITIDOS:

        return jsonify({
            "erro":
                "Estado inválido."
        }), 400


    estado_duque["estado"] = (
        novo_estado
    )


    estado_duque[
        "ultima_atualizacao"
    ] = datetime.now().isoformat(
        timespec="milliseconds"
    )


    print(
        "[DUQUE] "
        f"{novo_estado.upper()}"
    )


    return jsonify(
        estado_duque
    )


# ============================================================
# STATUS SIMPLES
# ============================================================

@app.route(
    "/status",
    methods=["GET"]
)
def status():

    return jsonify({
        "estado":
            estado_duque["estado"],

        "atividade":
            estado_duque["atividade"],

        "tarefa":
            estado_duque["tarefa"]
    })
